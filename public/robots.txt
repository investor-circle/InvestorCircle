# myinvestorcircle.com
#
# Public ideas and the stock pages that aggregate them are meant to be found.
# Member PROFILES are not: that was a deliberate decision. Profiles are real
# URLs now (/investor/name, served by the plain SPA shell — see index.html's
# canonical-tag comment) rather than HashRouter fragments a crawler could
# never see in the first place, which makes the rule below load-bearing, not
# a defensive backup: it is the only thing keeping them out of the index.

User-agent: *
Allow: /

# Not indexable, and not to be crawled.
Disallow: /investor/

# /search itself is crawlable (its own <meta name="robots"> is
# noindex,follow — see web-public/app/search/page.jsx) so its outbound
# links to /security/:symbol and /idea/:id get followed without the search
# results page itself being indexed as a destination.

Sitemap: https://myinvestorcircle.com/sitemap.xml
